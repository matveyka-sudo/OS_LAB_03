cmake_minimum_required(VERSION 3.16)
project(OS_Lab3_Variant10 C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(OS_Lab3_Variant10 main.c)
target_compile_options(OS_Lab3_Variant10 PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(OS_Lab3_Variant10 PRIVATE Threads::Threads)
if(UNIX AND NOT APPLE)
    target_link_libraries(OS_Lab3_Variant10 PRIVATE m)
endif()
enable_testing()
add_test(NAME generated_serial COMMAND OS_Lab3_Variant10 --size 100 --threads 1)
add_test(NAME generated_parallel COMMAND OS_Lab3_Variant10 --size 100 --threads 4)
add_test(NAME file_pivot COMMAND OS_Lab3_Variant10 --file ${CMAKE_CURRENT_SOURCE_DIR}/tests/pivot.txt --threads 3)
